control_baden = {
	is_shown = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
		exists = c:NGF
		c:BAD ?= {
			is_subject = no
		}
	}

	possible = {
		c:NGF = {
			is_unification_candidate = GER
			is_diplomatic_play_committed_participant = no
			NOT = { has_truce_with = c:BAD }
		}
	}

	when_taken = {
		if = {
			limit = {

			}
		}
	}

	ai_chance = {
		value = 0
	}
}
