ban_opium = {
	is_shown = {
		is_player = yes
		exists = market
		market.owner = root
		NOT = { is_banning_goods = g:opium }
		any_primary_culture = {
			has_cultural_obsession = opium
		}
	}

	possible = {
		is_player = yes
		NOT = { has_law_or_variant = law_type:law_free_trade }
		exists = market
	}

	when_taken = {
		add_banned_goods = g:opium
		every_primary_culture = {
			limit = {
				has_cultural_obsession = opium
			}
			remove_cultural_obsession = opium
		}
	}

    ai_chance = {
		value = 0
    }
}

play_as_random_tag = {
	is_shown = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
	}

	possible = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
	}

	when_taken = {
		random_subject_or_below = {
			limit = {
				country_rank >= rank_value:minor_power
				#liberty_desire > 50
			}
			save_scope_as = subject_target
		}
		if = {
			limit = { exists = scope:subject_target }
			play_as = scope:subject_target
		}
	}

	ai_chance = {
		value = 0
	}
}

dei_annex_split_puppet = {
	is_shown = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
		c:DEI ?= {
			any_subject_or_below = {
				count > 0
			}
		}
	}

	possible = {
		c:DEI = {
			#is_at_war = no
			is_diplomatic_play_committed_participant = no
			any_subject_or_below = {
				is_subject_type = subject_type_puppet
				liberty_desire < 25
				num_states <= 1
				capital = {
					is_split_state = yes
					state_region = {
						any_scope_state = { owner = c:DEI }
					}
				}
				NOT = { has_truce_with = c:DEI }
			}
		}
	}

	when_taken = {
		c:DEI = {
			random_subject_or_below = {
				limit = {
					is_subject_type = subject_type_puppet
					liberty_desire < 25
					num_states <= 1
					capital = {
						is_split_state = yes
						state_region = {
							any_scope_state = { owner = c:DEI }
						}
					}
					NOT = { has_truce_with = c:DEI }
				}
				save_scope_as = dei_annexation_target
			}
			if = {
				limit = { exists = scope:dei_annexation_target }
				create_diplomatic_play = {
					name = annex_subject
					target_country = scope:dei_annexation_target
					type = dp_annex_subject
				}
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
