ban_opium = {
	is_shown = {
		is_player = yes
		exists = market
		market.owner = root
		NOT = { is_banning_goods = g:opium }
		any_primary_culture = {
			has_cultural_obsession = opium
		}
	}

	possible = {
		is_player = yes
		NOT = { has_law_or_variant = law_type:law_free_trade }
		exists = market
	}

	when_taken = {
		add_banned_goods = g:opium
		every_primary_culture = {
			limit = {
				has_cultural_obsession = opium
			}
			remove_cultural_obsession = opium
		}
	}

    ai_chance = {
		value = 0
    }
}

support_lanfang_independence = {
	is_shown = {
		is_player = yes
		NOT = { c:LAN = THIS }
		c:LAN ?= {
			is_subject_of = c:CHI
		}
	}

	possible = {
		OR = {
			exists = c:STG
			c:LAN ?= {
				liberty_desire < 50
			}
		}
	}

	when_taken = {
		c:LAN ?= {
			add_liberty_desire = 45
		}
		c:CHI ?= {
			if = {
				limit = {
					NOT = {
						has_diplomatic_pact = {
							who = c:LAN
							type = decrease_payments
						}
					}
				}
				create_diplomatic_pact = {
					country = c:LAN
					type = decrease_payments
				}
			}
		}
		if = {
			limit = {
				c:LAN ?= {
					is_diplomatic_play_committed_participant = no
				}
				exists = c:STG
			}
			c:LAN ?= {
				create_diplomatic_play = {
					target_state = s:STATE_WEST_BORNEO.region_state:STG
					name = conquer_state
					type = dp_conquer_state

					add_initiator_backers = { c:BRU }
				}
				create_bidirectional_truce = { country = c:DEI months = 12 }
			}
		}
	}

	ai_chance = {
		value = 0
	}
}

			}
		}
	}

	possible = {
		is_player = yes
		c:TUR ?= {
			capital = { state_region = s:STATE_EASTERN_THRACE }

			set_capital = STATE_ANKARA
		}
	}

	ai_chance = {
		value = 100
	}
}

hk_treatyport = {
	is_shown = {
		is_player = yes
		year > 1840
		year < 1850
		c:CHI ?= {
			NOT = { has_truce_with = c:GBR }
			owns_entire_state_region = STATE_SHAOZHOU
		}
	}

	possible = {
		exists = c:GBR
		exists = c:CHI
		NOT = { c:GBR = THIS }
		c:CHI ?= {

		}

		# HK Treaty Port
		create_treaty = {

				{

					}
				}
			}
		}

			create_bidirectional_truce = { country = c:GBR months = 120 }

		value = 0
	}

trigger_serbia_turkey_war = {
	is_shown = {
		is_player = yes
		exists = c:SER
	}

	possible = {
		is_player = yes
		c:SER ?= {
			NOT = { has_truce_with = c:TUR }
			is_subject = no
		}

		s:STATE_EASTERN_SERBIA = {
			has_claim_by = c:SER
			state_region = {
				any_scope_state = { owner = c:SER }
			}
		}
	}

	when_taken = {
		c:SER ?= {
			create_diplomatic_play = {
				target_state = s:STATE_EASTERN_SERBIA.region_state:TUR
				name = return_state
				type = dp_return_state
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_EASTERN_SERBIA
				}
				if = {
					limit = {
						any_scope_building = {
							is_building_type = building_arms_industry
						}
					}
					remove_building = building_arms_industry
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_WESTERN_SERBIA
				}
				if = {
					limit = {
						NOT = {
							any_scope_building = {
								is_building_type = building_arms_industry
							}
						}
					}
					create_building = {
						building = building_arms_industry
						level = 1
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
