play_as_random_tag = {
	is_shown = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
	}

	possible = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
	}

	when_taken = {
		random_subject_or_below = {
			limit = {
				country_rank >= rank_value:minor_power
				#liberty_desire > 50
			}
			save_scope_as = subject_target
		}
		if = {
			limit = { exists = scope:subject_target }
			play_as = scope:subject_target
		}
	}

	ai_chance = {
		value = 0
	}
}

disband_all_basic_companies = {
	is_shown = {
		is_player = yes
	}

	possible = {
		is_player = yes
	}

	when_taken = {
		every_country = {
			limit = {
				is_player = no
				is_diplomatic_play_committed_participant = no
				#country_rank >= rank_value:minor_power
			}
			every_company = {
				if = {
					limit = {
						OR = {
							is_company_type = company_type:company_basic_forestry
							is_company_type = company_type:company_basic_wine_and_fruit
							is_company_type = company_type:company_basic_agriculture_1
							is_company_type = company_type:company_basic_agriculture_2
							is_company_type = company_type:company_basic_silk_and_dye
							is_company_type = company_type:company_basic_metal_mining
							is_company_type = company_type:company_basic_mineral_mining
							is_company_type = company_type:company_basic_paper
							is_company_type = company_type:company_basic_metalworks
							is_company_type = company_type:company_basic_fishing
							is_company_type = company_type:company_basic_colonial_plantations_1
							is_company_type = company_type:company_basic_colonial_plantations_2
							is_company_type = company_type:company_basic_fabrics
							is_company_type = company_type:company_basic_textiles
							is_company_type = company_type:company_basic_weapons
							#is_company_type = company_type:company_mantero_seta
						}
					}
					owner ?= {
						remove_company = prev.type
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
