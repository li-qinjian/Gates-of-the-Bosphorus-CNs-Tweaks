	}

	possible = {
		NOT = { c:BOH = THIS }
	}

	when_taken = {
		play_as = c:BOH
	}

	ai_chance = {
		value = 0
	}
}

play_as_hellas = {
	is_shown = {
		is_player = yes
		OR = {
			exists = c:GRE
			exists = c:BYZ
		}
	}

	possible = {
		OR = {
			NOT = { c:GRE = THIS }
			NOT = { c:BYZ = THIS }
		}
	}

	when_taken = {
		if = {
			limit = {
				exists = c:GRE
			}
			play_as = c:GRE
		}
		if = {
			limit = {
				exists = c:BYZ
			}
			play_as = c:BYZ
		}
	}
	ai_chance = {
		value = 0
	}
}

relocate_african_pops_to_capital = {
	is_shown = {
		is_player = yes
		NOT = { 		
			any_primary_culture = {
				has_discrimination_trait_group = heritage_group_african
			}
		}
	}

	possible = {
		has_law_or_variant = law_type:law_slavery_banned
		any_scope_state = {
			is_capital = no
			is_incorporated = yes
			is_homeland = cu:turkish
			any_scope_pop = {
				culture = {
					has_discrimination_trait_group = heritage_group_african
				}
			}
		}
	}

	when_taken = {
		every_scope_state = {
			limit = {
				is_capital = no
				is_incorporated = yes
				is_homeland = cu:turkish
			}
			every_scope_pop = {
				limit = {
					culture = {
						has_discrimination_trait_group = heritage_group_african
					}
				}
				move_pop = root.capital
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
