play_as_lanfang = {
	is_shown = {
		is_player = yes
		NOT = { c:LAN = THIS }
	}

	possible = {
		exists = c:LAN
	}

	when_taken = {
		play_as = c:LAN
	}

	ai_chance = {
		value = 0
	}
}

play_as_hellas = {
	is_shown = {
		is_player = yes
		NOT = { c:GRE = THIS }
		NOT = { c:BYZ = THIS }
	}

	possible = {
		OR = {
			exists = c:GRE
			exists = c:BYZ
		}
	}

	when_taken = {
		if = {
			limit = {
				exists = c:GRE
			}
			play_as = c:GRE
		}
		if = {
			limit = {
				exists = c:BYZ
			}
			play_as = c:BYZ
		}
	}
	ai_chance = {
		value = 0
	}
}

restore_eastern_roman_empire = {
	is_shown = {
		is_player = yes
		c:BYZ ?= THIS
		# any_scope_state = {
		# 	limit = {
		# 		has_claim_by = ROOT
		# 		NOT = {
		# 			OR = {
		# 				is_in_geographic_region = geographic_region_rumelia
		# 				is_in_geographic_region = geographic_region_anatolia
		# 			}
		# 		}
		# 	}
		# }
	}

	possible = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
	}

	when_taken = {
		every_scope_pop = {
			limit = {
				culture = cu:greek
			}
			change_pop_culture = {
				target = cu:rhomaios
				value = 1
			}
		}
		every_state_region = {
			limit = {
				region = sr:region_balkans
				any_scope_state = {
					owner = ROOT
					is_homeland = cu:turkish
					NOT = { has_claim_by = c:TUR }
					culture_percent_state = {
						target = cu:turkish
						value < 0.25
					}
				}
			}
			remove_homeland = cu:turkish
		}
		every_state = {
			limit = {
				has_claim_by = ROOT
				NOT = {
					OR = {
						is_in_geographic_region = geographic_region_rumelia
						is_in_geographic_region = geographic_region_anatolia
					}
				}
			}
			state_region = { remove_claim = ROOT }
		}
	}

	ai_chance = {
		value = 0
	}
}
