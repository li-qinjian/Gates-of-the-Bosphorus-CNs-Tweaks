ban_opium = {
	is_shown = {
		is_player = yes
		exists = market
		market.owner = root
		NOT = { is_banning_goods = g:opium }
		any_primary_culture = {
			has_cultural_obsession = opium
		}
	}

	possible = {
		is_player = yes
		NOT = { has_law_or_variant = law_type:law_free_trade }
		exists = market
	}

	when_taken = {
		add_banned_goods = g:opium
		every_primary_culture = {
			limit = {
				has_cultural_obsession = opium
			}
			remove_cultural_obsession = opium
		}
	}

    ai_chance = {
		value = 0
    }
}

lanfang_issue = {
	is_shown = {
		is_player = yes
		NOT = { c:LAN = THIS }
		c:LAN ?= {
			is_subject_of = c:CHI
		}
		c:CHI ?= { 
			country_has_primary_culture = cu:manchu 
		}
	}

	possible = {
		c:LAN ?= {
			liberty_desire < 50
		}
	}

	when_taken = {
		# Annex STG
		if = {
			limit = {
				exists = c:STG
			}
			c:LAN ?= {
				annex = c:STG
			}
		}
		# Add liberty desire to c:LAN
		c:LAN ?= {
			add_liberty_desire = 45
		}
		# Decrease payments diplomatic pact between c:CHI and c:LAN
		c:CHI ?= {
			if = {
				limit = {
					NOT = {
						has_diplomatic_pact = {
							who = c:LAN
							type = decrease_payments
						}
					}
				}
				create_diplomatic_pact = {
					country = c:LAN
					type = decrease_payments
				}
			}
		}
	}

	ai_chance = {
		value = 0
	}
}

chuangguan-dong = {
	is_shown = {
		is_player = yes
		c:CHI ?= { 
			NOT = { country_has_primary_culture = cu:han } 
		}
	}

	possible = {
		c:CHI ?= { 
			is_diplomatic_play_committed_participant = no
		}
	}

	when_taken = {
		c:CHI ?= {
			add_primary_culture = cu:han
			# Change the culture of Manchu peasants and laborers in incorporated states that are Manchu homeland but not Han homeland to Han culture
			every_scope_state = {
				limit = {
					is_incorporated = yes
					state_region = {
						is_homeland = cu:manchu
						NOT = { is_homeland = cu:han }
					}
				}
				every_scope_pop = {
					limit = {
						culture = cu:manchu
						OR = {
							is_pop_type = peasants
							is_pop_type = laborers
						}
					}
					change_pop_culture = {
						target = cu:han
						value = 0.75
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
