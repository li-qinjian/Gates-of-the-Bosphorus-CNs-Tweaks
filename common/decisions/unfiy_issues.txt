
dei_annex_split_puppet = {
	is_shown = {
		is_player = yes
		is_diplomatic_play_committed_participant = no
		c:DEI ?= {
			any_subject_or_below = {
				count > 0
			}
		}
	}

	possible = {
		c:DEI = {
			#is_at_war = no
			is_diplomatic_play_committed_participant = no
			any_subject_or_below = {
				is_subject_type = subject_type_puppet
				liberty_desire < 25
				num_states <= 1
				capital = {
					is_split_state = yes
					state_region = {
						any_scope_state = { owner = c:DEI }
					}
				}
				NOT = { has_truce_with = c:DEI }
			}
		}
	}

	when_taken = {
		c:DEI = {
			random_subject_or_below = {
				limit = {
					is_subject_type = subject_type_puppet
					liberty_desire < 25
					num_states <= 1
					capital = {
						is_split_state = yes
						state_region = {
							any_scope_state = { owner = c:DEI }
						}
					}
					NOT = { has_truce_with = c:DEI }
				}
				save_scope_as = dei_annexation_target
			}
			if = {
				limit = { exists = scope:dei_annexation_target }
				create_diplomatic_play = {
					name = annex_subject
					target_country = scope:dei_annexation_target
					type = dp_annex_subject
				}
			}
		}
	}

	ai_chance = {
		value = 0
	}
}
