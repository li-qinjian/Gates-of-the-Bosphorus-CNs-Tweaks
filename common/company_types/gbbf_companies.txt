		has_interest_marker_in_region = sr:region_balkans
	}

	attainable = {
		hidden_trigger = { # Hide attainable if it's always fulfilled by possible
			any_scope_state = {
				state_region = s:STATE_ATTICA
			}
		}
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_ATTICA
			is_incorporated = yes
			any_scope_building = {
				OR = {
					is_building_type = building_explosives_factory
					is_building_type = building_chemical_plant

				level >= 5
			}
		}
	}

	prosperity_modifier = {
		country_military_goods_cost_mult = -0.1
	}

	ai_will_do = {
		any_scope_state = {

		}
	}

	ai_construction_targets = {
		building_chemical_plant = {
			level = 5
			state_trigger = {
				state_region = s:STATE_ATTICA

		}
		building_explosives_factory = {

	}

	possible = {
		OR = {
			any_scope_state = {
				state_region = s:STATE_BOMBAY

			s:STATE_BOMBAY ?= {
				any_scope_state = {
					owner = {
						OR = {
							any_scope_treaty = {
								binds = ROOT
								any_scope_article = {
									has_type = trade_privilege
								}
							}
							top_overlord ?= {
								any_scope_treaty = {
									binds = ROOT
									any_scope_article = {
										has_type = trade_privilege
									}
								}
							}
						}
					}
				}
			}
		}
		any_scope_state = {
			has_state_trait = state_trait_burmese_teak

	prosperity_modifier = {
		building_tea_plantation_throughput_add = 0.1
		building_coffee_plantation_throughput_add = 0.1
	}

	ai_will_do = {
		OR = {
			any_scope_state = {
				state_region = s:STATE_BOMBAY
				#is_incorporated = yes
			}
			s:STATE_BOMBAY ?= {
				any_scope_state = {
					owner = {
						any_scope_treaty = {
							binds = root
							any_scope_article = {
								has_type = trade_privilege
							}
						}
					}
				}
			}
